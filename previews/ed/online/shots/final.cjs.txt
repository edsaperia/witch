const { chromium } = require("playwright");
(async () => {
  const b = await chromium.launch({ args: ["--use-gl=angle", "--use-angle=swiftshader", "--enable-unsafe-swiftshader", "--ignore-gpu-blocklist"] });
  const errs = [];
  const mk = async n => { const c = await b.newContext({ viewport: { width: 960, height: 540 } }); const p = await c.newPage(); p.on("console", m => { if (m.type() === "error") errs.push(`${n} console.error ${m.text()}`); }); p.on("pageerror", e => errs.push(`${n} pageerror ${e.message}`)); return p; };
  const s = await mk("solo");
  await s.goto("http://127.0.0.1:4178/?seed=123&quick=1");
  const ok = await s.waitForFunction(() => window.witch && window.witch.ready, null, { timeout: 240000, polling: 500 }).then(() => true).catch(() => false);
  console.log("solo ready", ok, await s.evaluate(() => document.getElementById("version")?.textContent));
  await s.screenshot({ path: "shots/02-ordinary-entry.png" }); await s.context().close();
  const h = await mk("host");
  await h.goto("http://127.0.0.1:4178/online/online.html?online=host&relay=ws%3A%2F%2Flocalhost%3A8787&seed=123&quick=1");
  await h.waitForFunction(() => window.witchOnline?.ready && !document.getElementById("invite").hidden, null, { timeout: 240000, polling: 500 });
  const g = await mk("guest"); await g.goto(await h.evaluate(() => document.getElementById("invite").href));
  await new Promise(r => setTimeout(r, 90000));
  console.log("after 90 s guest", JSON.stringify(await g.evaluate(() => ({ status: document.getElementById("status")?.textContent, ready: window.witchOnline?.ready, tick: window.witchOnline?.tick }))), "host", JSON.stringify(await h.evaluate(() => ({ status: document.getElementById("status").textContent, tick: window.witchOnline.tick, players: window.witchOnline.players.length }))));
  await h.screenshot({ path: "shots/03-host-with-guest-attached.png" }); await g.screenshot({ path: "shots/04-guest-at-90s-connecting.png" });
  console.log("errors", JSON.stringify(errs));
  await b.close();
})();
