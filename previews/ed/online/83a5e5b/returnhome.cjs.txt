const { chromium } = require("playwright"), path = require("path");
(async () => {
  const b = await chromium.launch({ args: ["--use-gl=angle", "--use-angle=swiftshader", "--enable-unsafe-swiftshader", "--ignore-gpu-blocklist"] });
  for (const n of [1, 2]) {
    const c = await b.newContext({ viewport: { width: 960, height: 540 } }); const p = await c.newPage(); const errs = [];
    p.on("pageerror", e => errs.push(e.message)); p.on("console", m => { if (m.type() === "error") errs.push(m.text()); });
    const t = Date.now();
    await p.goto("http://127.0.0.1:4178/online/online.html?online=guest&relay=ws%3A%2F%2Flocalhost%3A8787&seed=123&quick=1&room=1234567#invite=deadbeef");
    const back = await p.waitForFunction(() => !location.pathname.includes("/online/"), null, { timeout: 60000, polling: 200 }).then(() => true).catch(() => false);
    const url = p.url(), tb = (Date.now() - t) / 1000;
    const ok = back && await p.waitForFunction(() => window.witch && window.witch.ready, null, { timeout: 300000, polling: 500 }).then(() => true).catch(() => false);
    console.log(`try ${n}: returned ${back} in ${tb.toFixed(1)} s to ${url}; window.witch.ready ${ok} at ${((Date.now() - t) / 1000).toFixed(1)} s; errors ${JSON.stringify(errs).slice(0, 400)}`);
    await p.screenshot({ path: path.join("shots", `return-home-${n}.png`) });
    await c.close();
  }
  await b.close();
})();
